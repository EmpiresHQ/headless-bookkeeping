--- Table: account ---
CREATE TABLE account (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      code TEXT NOT NULL UNIQUE,
      name TEXT NOT NULL,
      type TEXT NOT NULL CHECK(type IN ('asset', 'liability', 'equity', 'revenue', 'expense')),
      currency TEXT,
      parent_id INTEGER REFERENCES account(id),
      is_system INTEGER NOT NULL DEFAULT 0
    )

--- Table: voucher ---
CREATE TABLE voucher (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      voucher_number TEXT NOT NULL UNIQUE,
      tax_point_date TEXT NOT NULL,
      posted_at INTEGER,
      previous_hash TEXT,
      reverses_id INTEGER REFERENCES voucher(id),
      corrects_object_type TEXT,
      corrects_object_id INTEGER,
      reason TEXT
    )

--- Table: voucher_line ---
CREATE TABLE voucher_line (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      voucher_id INTEGER NOT NULL REFERENCES voucher(id),
      account_id INTEGER NOT NULL REFERENCES account(id),
      amount INTEGER NOT NULL,
      currency TEXT NOT NULL,
      base_amount INTEGER NOT NULL,
      fx_rate REAL NOT NULL,
      vat_code TEXT,
      is_debit INTEGER NOT NULL
    )
